import streamlit as st
import pypdf
from openai import OpenAI

# Initialize OpenAI Client
client = OpenAI()

st.set_page_config(page_title="Doc Summarizer Chatbot", page_icon="📄")
st.title("📄 Document Summarizer & Assistant")

# Initialize Chat History
if "messages" not in st.session_state:
    st.session_state.messages = []
if "doc_text" not in st.session_state:
    st.session_state.doc_text = ""

# Sidebar: File Upload
with st.sidebar:
    st.header("Upload Document")
    uploaded_file = st.file_uploader("Upload a PDF or TXT file", type=["pdf", "txt"])

    if uploaded_file and not st.session_state.doc_text:
        with st.spinner("Processing file..."):
            if uploaded_file.type == "application/pdf":
                reader = pypdf.PdfReader(uploaded_file)
                st.session_state.doc_text = "\n".join([page.extract_text() for page in reader.pages if page.extract_text()])
            else:
                st.session_state.doc_text = uploaded_file.read().decode("utf-8")

            st.success("File uploaded successfully!")

# Quick Action: Generate Full Summary
if st.session_state.doc_text and st.button("Generate Document Summary"):
    with st.chat_message("assistant"):
        with st.spinner("Summarizing..."):
            # Truncate text if necessary to fit context window limits
            prompt = f"Summarize the following document concisely, highlighting key insights:\n\n{st.session_state.doc_text[:12000]}"
            response = client.chat.completions.create(
                model="gpt-4o-mini",
                messages=[{"role": "user", "content": prompt}],
            )
            summary = response.choices[0].message.content
            st.write(summary)
            st.session_state.messages.append({"role": "assistant", "content": summary})

# Render Existing Chat Messages
for msg in st.session_state.messages:
    with st.chat_message(msg["role"]):
        st.write(msg["content"])

# User Chat Input
if user_prompt := st.chat_input("Ask a question about the document..."):
    if not st.session_state.doc_text:
        st.error("Please upload a document first.")
    else:
        # Display user message
        st.chat_message("user").write(user_prompt)
        st.session_state.messages.append({"role": "user", "content": user_prompt})

        # Generate LLM response with document context
        with st.chat_message("assistant"):
            system_prompt = f"You are a helpful assistant. Use the provided document to answer questions accurately.\n\nDocument Context:\n{st.session_state.doc_text[:12000]}"

            chat_messages = [{"role": "system", "content": system_prompt}] + [
                {"role": m["role"], "content": m["content"]} for m in st.session_state.messages
            ]

            response = client.chat.completions.create(
                model="gpt-4o-mini",
                messages=chat_messages
            )
            bot_reply = response.choices[0].message.content
            st.write(bot_reply)
            st.session_state.messages.append({"role": "assistant", "content": bot_reply})
